cmake_minimum_required(VERSION 3.24)
project(pearl_sm120 LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)

if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 12.8)
  message(FATAL_ERROR "BLOCKER_TOOLCHAIN_CUDA_TOO_OLD")
endif()

add_library(pearl_sm120 STATIC sm120_backend.cu sm120_noise.cu sm120_dense_reference.cu sm120_fused_kernel.cu sm120_job_manager.cpp)
target_compile_features(pearl_sm120 PUBLIC cxx_std_17)
target_link_libraries(pearl_sm120 PUBLIC CUDA::cudart)
set_target_properties(pearl_sm120 PROPERTIES CUDA_ARCHITECTURES 120 CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
target_compile_options(pearl_sm120 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>)

include(CTest)
if(BUILD_TESTING)
  foreach(test_name sm120_reference_vectors sm120_transcript_parity sm120_blake3_parity sm120_live_replay)
    add_executable(${test_name} ../../tests/${test_name}.cpp)
    target_compile_features(${test_name} PRIVATE cxx_std_17)
    target_compile_options(${test_name} PRIVATE -UNDEBUG)
    target_include_directories(${test_name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(${test_name} PRIVATE pearl_sm120)
    add_test(NAME ${test_name} COMMAND ${test_name})
  endforeach()
  add_executable(sm120_submission_guard ../../tests/sm120_proof_parity.cpp)
  target_compile_features(sm120_submission_guard PRIVATE cxx_std_17)
  target_compile_options(sm120_submission_guard PRIVATE -UNDEBUG)
  target_include_directories(sm120_submission_guard PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(sm120_submission_guard PRIVATE pearl_sm120)
  add_test(NAME sm120_submission_guard COMMAND sm120_submission_guard)
  add_executable(sm120_dense_reference_cuda ../../tests/sm120_dense_reference_cuda.cu)
  target_link_libraries(sm120_dense_reference_cuda PRIVATE pearl_sm120)
  target_include_directories(sm120_dense_reference_cuda PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(sm120_dense_reference_cuda PRIVATE -UNDEBUG)
  add_test(NAME sm120_dense_reference_cuda COMMAND sm120_dense_reference_cuda)
  set_tests_properties(sm120_dense_reference_cuda PROPERTIES SKIP_RETURN_CODE 77)
endif()
